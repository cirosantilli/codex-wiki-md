$$
\boxed{
[K_i,K_j]=-\epsilon_{ijk}J_k,\qquad
[J_i,K_j]=\epsilon_{ijk}K_k,\qquad
[J_i,J_j]=\epsilon_{ijk}J_k}.
$$
