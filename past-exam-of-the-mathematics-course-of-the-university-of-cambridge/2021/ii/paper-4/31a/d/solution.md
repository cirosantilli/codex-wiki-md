$$
y_1\sim x^{-3/4}(1+1/(4x)+5/(32x^2))\cos(2\sqrt x+x^{-1/2}+x^{-3/2}/12),
$$



$$
y_2\sim x^{-3/4}(1+1/(4x)+5/(32x^2))\sin(2\sqrt x+x^{-1/2}+x^{-3/2}/12).
$$
